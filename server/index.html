import asyncio
from client import MedicalTriageEnv

# Connect to the world-class triage environment
env = await MedicalTriageEnv.from_url("https://current-space")

# Assess one of 3 difficulty levels (easy, medium, hard)
obs = await env.reset(task_id="hard")
print(obs.presentation)

# Execute triage action (ESI 1-5 + Department)
res = await env.step(esi_level=2, department="Emergency")
print(f"Score: {res.reward.value}")
